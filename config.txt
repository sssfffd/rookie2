# LogScope 설정. 이 파일 하나만 고치면 됩니다.
# 고치고 프로그램을 다시 켜면 바뀝니다. 지우면 기본값으로 돕니다.
#
#   name      창 제목과 왼쪽 위에 적히는 이름.
#             실행 파일 이름(out\<이름>.exe)도 이걸 따라갑니다 — 그건
#             build.bat 이 복사할 때 붙이므로 다시 빌드해야 바뀝니다.
#   analysis1 분석 화면의 이름. "이름 | 한 줄 설명" 으로 적습니다.
#   analysis2 세로줄(|) 뒤는 메인 화면 칸에 적히는 설명입니다 (없어도 됩니다).
#   analysis3
#
# 이 파일은 build.bat 이 out 폴더에 같이 넣어 줍니다. out\config.txt 를
# 바로 고쳐도 됩니다 (그때는 다시 빌드할 필요가 없습니다).
#
# 메모장으로 고쳐도 됩니다. UTF-8 로 저장하든 ANSI 로 저장하든 읽습니다.

name = LogScope

analysis1 = 로그 비교 | 이전 Log 파일과 이후 Log 파일을 비교합니다. 대시보드 · 그래프 · 히트맵.
analysis2 = 분석 2 | 아직 정해지지 않았습니다.
analysis3 = 분석 3 | 아직 정해지지 않았습니다.

# LogScope 설정. 이 파일 하나만 고치면 됩니다.
# 고치고 프로그램을 다시 켜면 바뀝니다. 지우면 기본값으로 돕니다.
#
#   name      창 제목과 왼쪽 위에 적히는 이름.
#             실행 파일 이름(out\<이름>.exe)도 이걸 따라갑니다 — 그건
#             build.bat 이 복사할 때 붙이므로 다시 빌드해야 바뀝니다.
#   fullname  긴 이름. 위 줄의 아랫줄(이름 아래)과 창 제목에 들어갑니다.
#             아랫줄에는 단추가 없어서 길게 적어도 윗줄을 밀지 않습니다.
#   analysis1 분석 화면의 이름. "이름 | 한 줄 설명" 으로 적습니다.
#   analysis2 세로줄(|) 뒤는 메인 화면 칸에 적히는 설명입니다 (없어도 됩니다).
#   analysis3
#
# 분석 2(DB)의 자리와 도구 경로는 여기 없습니다 — DB 화면과 설정 창에서
# 정하고, 프로그램이 settings.json 에 저장합니다.
#
# 이 파일은 build.bat 이 out 폴더에 같이 넣어 줍니다. out\config.txt 를
# 바로 고쳐도 됩니다 (그때는 다시 빌드할 필요가 없습니다).
#
# 메모장으로 고쳐도 됩니다. UTF-8 로 저장하든 ANSI 로 저장하든 읽습니다.

name = LogScope

# 긴 이름. 이름 아래 줄과 창 제목에 들어갑니다. 길이는 마음대로.
fullname = Log 비교 · 분석 도구

analysis1 = 로그 비교 | 이전 Log 파일과 이후 Log 파일을 비교합니다. 대시보드 · 그래프 · 히트맵.
analysis2 = DB 분석 | 두 DB (.sql / .csv) 의 값이 달라진 항목을 찾습니다.
analysis3 = 분석 3 | 아직 정해지지 않았습니다.

# 분석 1 공정 일치율 → 판정 경계 (1.92, 1.95 에 95 · 80). good 이상 = 정상, warn 이상 = 주의, 그 아래 = 이상.
score.good = 95
score.warn = 80
# 오차 크기 반영 (1.93): 이전 값보다 이 % 만큼 벗어날 때마다 그 오차 칸을 한 칸 더 (최대 3 배).
score.size = 50
# IO 하나가 공정 일치율에 미치는 영향 상한 % (2.00). 값 차이가 아주 큰 IO 하나가 점수를 통째로
# 끌어내리지 않게 — 그 IO 는 목록 · 판정 (이상) 에 그대로 보이고, 게이지 아래에 몇 개인지 적힙니다. 0 이면 상한 없음.
score.iocap = 15

# 최종 판정 (2.02) = 결과가 있는 분석의 가중 종합 점수. 분석 1 · 2 · 3 가중치 — AHP 쌍대비교 (분석 1 이 분석 2 의 2 배,
# 분석 3 의 3 배, 분석 2 가 분석 3 의 2 배 중요) 로 정한 54 · 30 · 16. 분석 1 은 공정 일치율, 분석 2 · 3 은 판정 구간의
# 가운데 점수 (정상 97.5 · 주의 87.5 · 이상 40) 로 넣고, 종합 점수를 위 경계 (95 · 80) 로 판정합니다.
final.weights = 54, 30, 16
# 1 = 넉아웃: 분석 1 이 이상이면 최종 이상, 다른 분석이 이상이면 최종은 적어도 주의. 0 = 점수만.
final.gate = 1

# 메인 대시보드 "차이 IO" 칸의 이름 (1.95).
label.diff = 차이 IO
